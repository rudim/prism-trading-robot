MqlCalendarValue Structure (from MT5 API):
──────────────────────────────────────────
  long                event_id;        // Event ID
  datetime            time;            // Event time
  datetime            period;          // Period for which published
  int                 revision;        // Revision indicator
  long                actual_value;    // Actual value (if available)
  long                prev_value;      // Previous value
  long                revised_prev_value; // Revised previous value
  long                forecast_value;  // Forecast value
  ENUM_CALENDAR_EVENT_IMPACT impact_type; // Impact level

MqlCalendarEvent Structure (from MT5 API):
───────────────────────────────────────────
  ulong               id;              // Event ID
  ENUM_CALENDAR_EVENT_TYPE type;       // Event type
  ENUM_CALENDAR_EVENT_SECTOR sector;   // Economic sector
  ENUM_CALENDAR_EVENT_FREQUENCY frequency; // Release frequency
  ENUM_CALENDAR_EVENT_TIMEMODE time_mode;  // Time mode
  ulong               country_id;      // Country ID
  ENUM_CALENDAR_EVENT_UNIT unit;       // Value unit
  ENUM_CALENDAR_EVENT_IMPORTANCE importance; // Importance level
  ENUM_CALENDAR_EVENT_MULTIPLIER multiplier; // Value multiplier
  uint                digits;          // Decimal places
  string              source_url;      // Source URL
  string              event_code;      // Event code
  string              name;            // Event name

MqlCalendarCountry Structure (from MT5 API):
─────────────────────────────────────────────
  ulong               id;              // Country ID
  string              name;            // Country name
  string              code;            // Country code (ISO 3166-1 alpha-2)
  string              currency;        // Currency code (USD, EUR, etc.)
  string              currency_symbol; // Currency symbol ($, €, etc.)
  string              url_name;        // URL-friendly country name

CalendarEvent Structure (Custom):
──────────────────────────────────
  datetime            eventTime;       // When event occurs
  string              currency;        // Currency code (extracted from country)
  string              eventName;       // Event description (from event.name)
  ENUM_CALENDAR_EVENT_IMPORTANCE importance; // Impact level
  bool                isPast;          // True if already occurred
